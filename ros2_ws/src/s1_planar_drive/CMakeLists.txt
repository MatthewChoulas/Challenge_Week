cmake_minimum_required(VERSION 3.16)
project(s1_planar_drive)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz_transport_vendor REQUIRED)
find_package(gz_common_vendor REQUIRED)
find_package(gz-sim8 REQUIRED COMPONENTS gui)
find_package(gz-common5 REQUIRED COMPONENTS graphics)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-transport13 REQUIRED)
add_library(s1_planar_drive SHARED src/PlanarDrive.cc)
target_compile_features(s1_planar_drive PRIVATE cxx_std_17)
target_link_libraries(s1_planar_drive PRIVATE gz-sim8::gz-sim8
  gz-plugin2::register gz-transport13::gz-transport13)
install(TARGETS s1_planar_drive LIBRARY DESTINATION lib)
add_library(s1_terrain_drive SHARED src/TerrainDrive.cc)
target_compile_features(s1_terrain_drive PRIVATE cxx_std_17)
target_link_libraries(s1_terrain_drive PRIVATE gz-sim8::gz-sim8
  gz-plugin2::register gz-transport13::gz-transport13 gz-common5::graphics)
install(TARGETS s1_terrain_drive LIBRARY DESTINATION lib)
add_executable(s1_gazebo_gui src/GazeboGui.cc)
target_compile_features(s1_gazebo_gui PRIVATE cxx_std_17)
target_link_libraries(s1_gazebo_gui PRIVATE gz-sim8::gz-sim8-gui)
install(TARGETS s1_gazebo_gui RUNTIME DESTINATION bin)
ament_environment_hooks(hooks/planar_drive.dsv)
ament_package()
